## Implementation Approach

Build PlaybookLM as a new tool at `/en-gb/playbooklm/` following the infographic generator as a reference. The infographic generator is the closest existing tool — PlaybookLM wraps the same `sg-llm-request` generation flow but adds a multi-step pipeline on top.

**Principle:** Every file should be under 200 lines. Split by responsibility, not by convenience.

### 1.3 Create `pipeline.js` (~180 lines)

Central state management. This is the pipeline engine — all step logic routes through here.

```javascript
export class PipelineState {
    constructor() {
        this.sources = [];
        this.presentationDoc = null;
        this.slideBriefs = [];
        this.slideResults = [];
        this.apiKey = null;
        this.model = 'google/gemini-3.1-flash-image-preview';
        this.textModel = 'anthropic/claude-3.5-sonnet';  // for Steps 2-3
        this.systemPrompt = DEFAULT_SYSTEM_PROMPT;
        this.brandingPrompt = null;
        this.connected = false;
        this._activeGenerations = new Map();  // callId → { resolve, reject }
        this._listeners = [];  // UI update callbacks
    }

    // Step 1
    async loadSources(files) { /* read files, detect role (content vs branding) */ }
    getSources() { return this.sources; }
    removeSource(index) { this.sources.splice(index, 1); this._notify(); }

    // Step 2
    async generatePresentation({ model } = {}) {
        const m = model || this.textModel;
        const sourceText = this.sources
            .filter(s => s.role === 'content')
            .map(s => `## ${s.name}\n${s.content}`)
            .join('\n\n---\n\n');

        // Build prompt
        const prompt = STEP2_PROMPT.replace('${sourceContents}', sourceText);

        // Call LLM via sg-llm-request pattern
        const result = await this._callTextLLM(prompt, m);
        this.presentationDoc = result;
        this._emit('tool:presentation:complete', { length: result.length });
        this._notify();
        return result;
    }
    getPresentation() { return this.presentationDoc; }
    setPresentation(text) { this.presentationDoc = text; this._notify(); }

    // Step 3
    async generateSlideBriefs({ count = 8, model } = {}) {
        const m = model || this.textModel;
        const prompt = STEP3_PROMPT
            .replace('${numSlides}', count)
            .replace('${presentationDoc}', this.presentationDoc);

        const result = await this._callTextLLM(prompt, m);
        // Parse JSON from LLM response
        this.slideBriefs = this._parseSlideJSON(result);
        this._emit('tool:briefs:complete', { count: this.slideBriefs.length });
        this._notify();
        return this.slideBriefs;
    }
    getSlideBriefs() { return this.slideBriefs; }
    setSlideBrief(index, brief) { this.slideBriefs[index] = { ...this.slideBriefs[index], ...brief }; this._notify(); }
    addSlideBrief(brief) { this.slideBriefs.push(brief); this._notify(); }
    removeSlideBrief(index) { this.slideBriefs.splice(index, 1); this._notify(); }

    // Step 4
    async generateSlide({ index, model, template } = {}) {
        const brief = this.slideBriefs[index];
        if (!brief) throw new Error(`No brief at index ${index}`);

        const m = model || this.model;
        const t = template || brief.template;

        // Compose three-layer prompt
        const systemMsg = this._buildSystemPrompt(t);
        const userMsg = this._buildPositioningPrompt() + '\n\n' + brief.prompt;

        // Call image LLM
        const result = await this._callImageLLM(systemMsg, userMsg, m);

        // Store result
        if (!this.slideResults[index]) {
            this.slideResults[index] = { index, generations: [], selectedGeneration: 0 };
        }
        this.slideResults[index].generations.push(result);
        this.slideResults[index].selectedGeneration = this.slideResults[index].generations.length - 1;

        this._emit('tool:slide:complete', { index, ...result });
        this._notify();
        return result;
    }

    async generateAllSlides({ model, parallel = false } = {}) {
        const results = [];
        if (parallel) {
            const promises = this.slideBriefs.map((_, i) =>
                this.generateSlide({ index: i, model }));
            return await Promise.all(promises);
        } else {
            for (let i = 0; i < this.slideBriefs.length; i++) {
                results.push(await this.generateSlide({ index: i, model }));
            }
            return results;
        }
    }
    getSlideResults() { return this.slideResults; }

    // Step 5
    async exportDeck({ format = 'pdf' } = {}) { /* jsPDF assembly */ }

    // Helpers
    _buildSystemPrompt(template) { /* default + branding + template-specific */ }
    _buildPositioningPrompt() { /* presentation doc + "slide N of M" context */ }
    async _callTextLLM(prompt, model) { /* sg-llm-request for text models */ }
    async _callImageLLM(system, user, model) { /* sg-llm-request for image models */ }
    _parseSlideJSON(text) { /* extract JSON array from LLM response */ }
    _emit(event, detail) { window.dispatchEvent(new CustomEvent(event, { detail })); }
    _notify() { this._listeners.forEach(fn => fn(this)); }
    onChange(fn) { this._listeners.push(fn); }
    getFullState() { /* return snapshot of entire pipeline */ }
    getStatus() { /* return { step1: 'complete', step2: 'complete', step3: 'pending', ... } */ }
    stop() { /* cancel all active generations */ }
}
```

```javascript
export function buildSourcesPanel(container, pipeline) {
    const panel = document.createElement('div');
    panel.className = 'pipeline-step';
    panel.innerHTML = `
        <h3>Step 1: Sources</h3>
        <div class="drop-zone" data-automation="source-dropzone">
            Drop files here or click to browse
            <input type="file" multiple accept=".txt,.md,.markdown,.pdf,.csv,.json,.html">
        </div>
        <div class="source-list" data-automation="source-list"></div>
    `;

    const dropZone = panel.querySelector('.drop-zone');
    const fileInput = panel.querySelector('input[type="file"]');
    const sourceList = panel.querySelector('.source-list');

    // Drag-drop handling
    dropZone.addEventListener('dragover', e => { e.preventDefault(); dropZone.classList.add('drag-over'); });
    dropZone.addEventListener('dragleave', () => dropZone.classList.remove('drag-over'));
    dropZone.addEventListener('drop', async e => {
        e.preventDefault();
        dropZone.classList.remove('drag-over');
        await pipeline.loadSources([...e.dataTransfer.files]);
    });

    dropZone.addEventListener('click', () => fileInput.click());
    fileInput.addEventListener('change', async () => {
        await pipeline.loadSources([...fileInput.files]);
        fileInput.value = '';
    });

    // Update UI when sources change
    pipeline.onChange(() => {
        sourceList.innerHTML = pipeline.getSources().map((s, i) => `
            <div class="source-item">
                <span class="source-name">${s.name}</span>
                <span class="source-role">${s.role}</span>
                <span class="source-size">${(s.content.length / 1024).toFixed(1)} KB</span>
                <button data-remove="${i}">✕</button>
            </div>
        `).join('');

        sourceList.querySelectorAll('[data-remove]').forEach(btn => {
            btn.addEventListener('click', () => pipeline.removeSource(parseInt(btn.dataset.remove)));
        });
    });

    container.appendChild(panel);
}
```

For MVP, support text/markdown file reading via `FileReader.readAsText()`. PDF support deferred to Phase 2 (use `pdf.js` or multimodal model).

### Key: LLM Call for Text (not image)

The presentation extraction uses a TEXT model, not an image model. Create a helper in `pipeline.js`:

```javascript
async _callTextLLM(prompt, model) {
    // Create a temporary sg-llm-request for text-only calls
    // OR use fetch() directly to OpenRouter:
    const res = await fetch('https://openrouter.ai/api/v1/chat/completions', {
        method: 'POST',
        headers: {
            'Authorization': `Bearer ${this.apiKey}`,
            'Content-Type': 'application/json',
        },
        body: JSON.stringify({
            model,
            messages: [{ role: 'user', content: prompt }],
            max_tokens: 4096,
        }),
    });
    const data = await res.json();
    return data.choices?.[0]?.message?.content || '';
}
```

Using `fetch()` directly is simpler than wiring `sg-llm-request` for text-only calls. `sg-llm-request` is designed for the streaming/image flow — text extraction is a one-shot call.

**Critical: Three-layer prompt composition.**

```javascript
_buildSystemPrompt(template) {
    let sys = DEFAULT_INFOGRAPHIC_SYSTEM_PROMPT;
    if (this.brandingPrompt) sys += '\n\n' + this.brandingPrompt;
    // Template-specific additions
    const tmpl = TEMPLATES.find(t => t.id === template);
    if (tmpl?.systemAddition) sys += '\n\n' + tmpl.systemAddition;
    return sys;
}

_buildPositioningPrompt() {
    if (!this.presentationDoc) return '';
    return `CONTEXT: This is part of a ${this.slideBriefs.length}-slide presentation deck.\n\n` +
           `PRESENTATION OVERVIEW:\n${this.presentationDoc}\n\n` +
           `Ensure this slide is visually consistent with the rest of the deck. ` +
           `Use the same colour scheme, style, and level of detail throughout.\n\n` +
           `SLIDE CONTENT:\n`;
}
```

For the image LLM call, use `sg-llm-request`:

```javascript
async _callImageLLM(systemPrompt, userPrompt, model) {
    return new Promise((resolve, reject) => {
        const callId = crypto.randomUUID();
        const miniBus = document.createElement('div');
        miniBus.setAttribute('data-llm-bus', '');
        document.getElementById('llm-bus').appendChild(miniBus);

        const llmReq = document.createElement('sg-llm-request');
        miniBus.appendChild(llmReq);

        // Fire connected event on miniBus
        miniBus.dispatchEvent(new CustomEvent('llm:connected', {
            detail: { apiKey: this.apiKey, model, provider: 'openrouter' }
        }));

        const t0 = Date.now();

        // Listen for completion
        miniBus.addEventListener('llm:request:complete', ev => {
            const src = extractImageSrc(ev.detail);
            const genId = ev.detail?.rawResponse?.id ?? null;
            const result = {
                callId,
                generationId: genId,
                model,
                duration: (Date.now() - t0) / 1000,
                imageSrc: src || null,
            };
            miniBus.remove();
            resolve(result);
        }, { once: true });

        miniBus.addEventListener('llm:request:error', ev => {
            miniBus.remove();
            reject(new Error(ev.detail?.error || 'Generation failed'));
        }, { once: true });

        // Fire the send event
        miniBus.dispatchEvent(new CustomEvent('llm:send', {
            detail: {
                systemPrompt,
                messages: [{ role: 'user', content: userPrompt }],
                model,
            }
        }));
    });
}
```

**Note:** Check the actual `sg-llm-request` event names in the infographic generator's `send.js` — they use `SGL_LLM.SEND`, `SGL_LLM.REQUEST_COMPLETE`, etc. from the bus events. Match those exactly.

Each generated slide creates a new tab in the right preview panel showing the infographic image, brief text, cost/duration, and action buttons (Regenerate, Try Different Model).

```javascript
async exportPDF() {
    const { jsPDF } = window.jspdf;
    const slides = this.slideResults.filter(s => s?.generations?.length > 0);

    // Get dimensions from first image
    const firstImg = new Image();
    firstImg.src = slides[0].generations[slides[0].selectedGeneration].imageSrc;
    await new Promise(r => { firstImg.onload = r; });

    const W = firstImg.naturalWidth;
    const H = firstImg.naturalHeight;
    const pdf = new jsPDF({ orientation: W > H ? 'landscape' : 'portrait', unit: 'px', format: [W, H] });

    for (let i = 0; i < slides.length; i++) {
        const gen = slides[i].generations[slides[i].selectedGeneration];
        if (i > 0) pdf.addPage([W, H]);
        pdf.addImage(gen.imageSrc, 'PNG', 0, 0, W, H);
    }

    pdf.save('PlaybookLM-Deck.pdf');
}
```

## Things to Watch Out For

1. **LLM JSON parsing:** The Step 3 LLM call returns a JSON array embedded in a text response. The parser must handle markdown fences (` ```json ... ``` `), preamble text, and trailing commentary. Strip everything outside `[...]`.

2. **`sg-llm-request` event names:** Check `constants.js` in infographic-gen for the exact bus event names (`SGL_LLM.SEND`, `SGL_LLM.REQUEST_COMPLETE`, etc.). Don't guess — read the source.

3. **System prompt composition:** The branding source document should be extracted during `loadSources()` and stored separately as `pipeline.brandingPrompt`. It goes into Layer 1 of the three-layer prompt. Detection: if a source filename contains "brand" or the user tags it as branding.

4. **Tab management in right panel:** Each slide gets its own tab in the stack layout. Use `previewLayout.addTabToStack()` per infographic-gen pattern. Tab titles should be "Slide 1: {title}", etc.

5. **Cost tracking:** Every `_callImageLLM()` call returns a `generationId`. Collect these in `pipeline.slideResults[].generations[].generationId`. The export step can batch-fetch costs from OpenRouter and include them in the generation log.

6. **Text model vs image model:** Steps 2-3 use a TEXT model (Claude, Gemini text). Step 4 uses an IMAGE model (gemini-3.1-flash-image-preview). Make sure the model picker or pipeline config allows selecting both.
